GRID_SIZE = 4

if grid is None:
    return FailureResult(code="INVALID_SIZE", message="Grid must be 4x4.")
if len(grid) != GRID_SIZE or any(len(row) != GRID_SIZE for row in grid):
    return FailureResult(code="INVALID_SIZE", message="Grid must be 4x4.")
raise NotImplementedError("AC-FR-01-01: input validation not implemented")
